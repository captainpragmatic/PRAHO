notifications.EmailLog          # append-only email delivery records
notifications.EmailAttachment     # a reference (source type, id, hash) owned by its EmailLog, which is audited
notifications.PendingDeliveryEvent  # short-lived buffer of verified provider callbacks; applied to its EmailLog (audited there) or expired
notifications.UnsubscribeToken  # ephemeral GDPR unsubscribe tokens (short-lived, append-only)
orders.OrderStatusHistory       # order status change log
users.UserLoginLog              # security audit trail for login attempts
